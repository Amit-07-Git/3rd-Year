def print_table(table, cols):
    for i in range(0, len(table), cols):
        print(" ".join(f"{num:02d}" for num in table[i:i+cols]))

def initial_permutation():
    table = []
    row_starts = [58, 60, 62, 64, 57, 59, 61, 63]
    for i in row_starts:
        for j in range(8):
            table.append(i - (j * 8))
    return table

def get_expansion_table():

    table = []
