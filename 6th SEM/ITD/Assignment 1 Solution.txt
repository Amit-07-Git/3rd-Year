1.

create table branch (
    branch_code varchar(5),
    branch_name varchar(30),
    city varchar(20)
);

create table customer (
    cust_no varchar(5),
    cust_name varchar(30),
    address varchar(40),
    phone numeric(10)
);

create table account (
    acc_no varchar(10),
    cust_no varchar(5),
    branch_code varchar(5),
    acc_type varchar(10),
    balance numeric(10)
);

create table loan (
    loan_no varchar(10),
    cust_no varchar(5),
    branch_code varchar(5),
    amount numeric(10)
);

create table transaction_details (
    trans_id varchar(10),
    acc_no varchar(10),
    trans_type varchar(10),
    trans_amount numeric(10),
    trans_date date
);
